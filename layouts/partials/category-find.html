{{- /* Поиск узла категории по slug. Возвращает узел или false. В Hugo 0.150 — только один return на шаблон. */ -}} {{- $slug := .slug -}} {{- $found := false -}} {{- range site.Data.categories -}} {{- if eq .slug $slug -}} {{- $found = . -}} {{- else if .children -}} {{- range .children -}} {{- if eq .slug $slug -}} {{- $found = . -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- return $found -}}
